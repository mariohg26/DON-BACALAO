const USUARIOS = [
  { nombre: 'Encargado', emoji: '👨‍💼', pin: '1234' },
  { nombre: 'Almacen',   emoji: '📦',   pin: '0000' },

let usuarioActual=null, pinIntroducido='', usuarioSeleccionado=null;

function initLogin(){
  const ul=document.getElementById('userList');
  ul.innerHTML=USUARIOS.map((u,i)=>`
    <button class="user-btn" onclick="selectUser(${i})">
      <div class="user-avatar">${u.emoji}</div>
      <span>${u.nombre}</span>
    </button>`).join('');
}
function selectUser(i){
  usuarioSeleccionado=USUARIOS[i];pinIntroducido='';
  document.getElementById('loginTitle').textContent='PIN de '+usuarioSeleccionado.nombre;
  document.getElementById('userSelectStep').style.display='none';
  document.getElementById('pinStep').style.display='block';
  updateDots();
}
function backUser(){
  document.getElementById('loginTitle').textContent='Selecciona usuario';
  document.getElementById('userSelectStep').style.display='block';
  document.getElementById('pinStep').style.display='none';
  document.getElementById('pinError').textContent='';
  pinIntroducido='';
}
function pinPress(d){
  if(pinIntroducido.length>=4)return;
  pinIntroducido+=d; updateDots();
  if(pinIntroducido.length===4)setTimeout(checkPin,150);
}
function pinDel(){pinIntroducido=pinIntroducido.slice(0,-1);updateDots();}
function updateDots(){for(let i=0;i<4;i++)document.getElementById('d'+i).classList.toggle('filled',i<pinIntroducido.length);}
function checkPin(){
  if(pinIntroducido===usuarioSeleccionado.pin){
    usuarioActual=usuarioSeleccionado;
    document.getElementById('loginScreen').style.display='none';
    document.getElementById('mainHeader').style.display='flex';
    document.getElementById('mainContainer').style.display='block';
    document.getElementById('mainNav').style.display='flex';
    document.getElementById('userBadge').textContent=usuarioActual.emoji+' '+usuarioActual.nombre;
    if(!document.getElementById('responsableLabel').value)
      document.getElementById('responsableLabel').value=usuarioActual.nombre;
  } else {
    document.getElementById('pinError').textContent='PIN incorrecto. Intentalo de nuevo.';
    pinIntroducido=''; updateDots();
    setTimeout(()=>document.getElementById('pinError').textContent='',2000);
  }
}
function cerrarSesion(){
  if(confirm('Cerrar sesion?')){
    usuarioActual=null;pinIntroducido='';usuarioSeleccionado=null;
    document.getElementById('loginScreen').style.display='flex';
    document.getElementById('mainHeader').style.display='none';
    document.getElementById('mainContainer').style.display='none';
    document.getElementById('mainNav').style.display='none';
    document.getElementById('loginTitle').textContent='Selecciona usuario';
    document.getElementById('userSelectStep').style.display='block';
    document.getElementById('pinStep').style.display='none';
    updateDots();
  }
}

  try{localStorage.setItem('donbacalao',JSON.stringify({p:productos,m:movimientos,h:historial,n:notas}));}catch(e){}
}
function loadData(){
  try{
    const d=JSON.parse(localStorage.getItem('donbacalao')||'{}');
    productos=d.p||[]; movimientos=d.m||[]; historial=d.h||[]; notas=d.n||[];
  }catch(e){}
}

document.getElementById('semanaLabel').value='Semana '+getWeekNum()+' - '+hoy.toLocaleDateString('es-ES',{month:'long',year:'numeric'});

loadData();
if(!productos.length) initCatalogo();
initLogin();
renderAll();

function initCatalogo(){
  productos=[];
  CATALOGO.forEach(({cat,prods})=>prods.forEach(p=>productos.push({id:uid(),nombre:p.n,cantidad:'',unidad:p.u,categoria:cat,bloqueado:false,fijo:true,nota:''})));
}
function uid(){return Date.now()+'-'+Math.random().toString(36).slice(2);}

function showSection(s,btn){
  document.querySelectorAll('.section').forEach(el=>el.classList.remove('active'));
  document.querySelectorAll('.bnav-btn').forEach(b=>b.classList.remove('active'));
  document.getElementById('sec-'+s).classList.add('active');
  if(btn)btn.classList.add('active');
  if(s==='alertas')renderAlertas();
  if(s==='movimientos'){renderMovSelect();renderMovimientos();}
  if(s==='notas')renderNotas();
}
function toggleAddExtra(){
  const p=document.getElementById('panelExtra');
  p.style.display=p.style.display==='none'?'block':'none';
}
function renderAll(){renderCatalogo();renderCatFilter();renderMovSelect();renderHistorial();}

function renderCatFilter(){
  const cats=[...new Set(productos.map(p=>p.categoria))];
  const el=document.getElementById('catFilter');
  el.innerHTML=`<button class="cat-chip ${filtroActivo===''?'active':''}" onclick="setFiltro('',this)">Todos</button>`+
    cats.map(c=>`<button class="cat-chip ${filtroActivo===c?'active':''}" onclick="setFiltro('${c.replace(/'/g,"\\'")}',this)">${c}</button>`).join('');
}
function setFiltro(cat,btn){filtroActivo=cat;document.querySelectorAll('.cat-chip').forEach(b=>b.classList.remove('active'));btn.classList.add('active');renderCatalogo();}

function renderCatalogo(){
  const busq=(document.getElementById('searchInput').value||'').toLowerCase();
  const cats=[...new Set(productos.map(p=>p.categoria))];
  const container=document.getElementById('catalogoContainer');
  container.innerHTML='';
  cats.forEach(cat=>{
    if(filtroActivo&&filtroActivo!==cat)return;
    const items=productos.filter(p=>p.categoria===cat&&(!busq||p.nombre.toLowerCase().includes(busq)));
    if(!items.length)return;
    const wrap=document.createElement('div');
    wrap.innerHTML=`<div class="cat-label">${cat}</div>`;
    items.forEach(p=>{
      const gi=productos.indexOf(p);
      const div=document.createElement('div');
      div.className='prod-card'+(p.bloqueado?' bloqueado':'');
      div.innerHTML=`
        <div class="pname">${p.nombre}</div>
        <div class="pmeta">${p.unidad||''}&nbsp;${p.bloqueado?'<span class="badge badge-blocked">BLOQUEADO</span>':'<span class="badge badge-green">OK</span>'}</div>
        <div class="qty-row">
          <input type="text" value="${esc(p.cantidad)}" placeholder="Cantidad"
            class="${p.cantidad?'filled':''}"
            onchange="productos[${gi}].cantidad=this.value;saveData();this.className=this.value?'filled':''">
          <button class="quick-btn" onclick="sumarKg(${gi},5)">+5kg</button>
          <button class="quick-btn" onclick="sumarKg(${gi},10)">+10kg</button>
        </div>
        ${p.nota?`<div class="prod-note">📌 ${esc(p.nota)}</div>`:''}
        <div class="prod-actions">
          <button class="btn-sm btn-green" style="flex:1" onclick="openModal(${gi},'entrada')">📥</button>
          <button class="btn-sm btn-warn" style="flex:1" onclick="openModal(${gi},'salida')">📤</button>
          <button class="btn-sm btn-ghost" style="flex:1" onclick="editNota(${gi})">📝</button>
          <button class="btn-sm ${p.bloqueado?'btn-primary':'btn-danger'}" onclick="toggleBloqueo(${gi})">${p.bloqueado?'✅':'⛔'}</button>
          ${!p.fijo?`<button class="btn-sm btn-ghost" onclick="deleteProduct(${gi})">🗑</button>`:''}
        </div>`;
      wrap.appendChild(div);
    });
    container.appendChild(wrap);
  });
}

function sumarKg(gi,kg){
  const p=productos[gi];
  p.cantidad=String((parseFloat(p.cantidad)||0)+kg);
  saveData();renderCatalogo();
}
function editNota(gi){
  const p=productos[gi];
  const n=prompt('Nota para "'+p.nombre+'":\n(deja vacio para borrar)',p.nota||'');
  if(n===null)return;
  p.nota=n.trim();saveData();renderCatalogo();
}
function toggleBloqueo(i){productos[i].bloqueado=!productos[i].bloqueado;saveData();renderAll();}
function deleteProduct(i){if(confirm('Eliminar "'+productos[i].nombre+'"?')){productos.splice(i,1);saveData();renderAll();}}
function resetCantidades(){if(confirm('Resetear todas las cantidades?')){productos.forEach(p=>p.cantidad='');saveData();renderAll();}}
function addExtra(){
  const n=document.getElementById('inpNombre').value.trim();
  if(!n){alert('Introduce un nombre.');return;}
  productos.push({id:uid(),nombre:n,cantidad:document.getElementById('inpCantidad').value.trim(),unidad:document.getElementById('inpUnidad').value.trim()||'ud',categoria:document.getElementById('inpCategoria').value||'Extra',bloqueado:false,fijo:false,nota:''});
  ['inpNombre','inpCantidad','inpUnidad'].forEach(id=>document.getElementById(id).value='');
  saveData();renderAll();document.getElementById('panelExtra').style.display='none';
}

function openModal(idx,tipo){
  modalData={idx,tipo};
  document.getElementById('modalTitle').textContent=(tipo==='entrada'?'📥 Entrada':'📤 Salida')+' — '+productos[idx].nombre;
  ['modalCantidad','modalMotivo','modalResp'].forEach(id=>document.getElementById(id).value='');
  document.getElementById('modalOverlay').classList.add('open');
  setTimeout(()=>document.getElementById('modalCantidad').focus(),200);
}
function closeModal(){document.getElementById('modalOverlay').classList.remove('open');}
function confirmModal(){
  const cant=document.getElementById('modalCantidad').value.trim();
  if(!cant){alert('Introduce una cantidad.');return;}
  aplicarMov(modalData.idx,modalData.tipo,cant,document.getElementById('modalMotivo').value.trim(),document.getElementById('modalResp').value.trim());
  closeModal();
}
function renderMovSelect(){
  const sel=document.getElementById('movProducto');const cur=sel.value;
  sel.innerHTML='<option value="">-- Selecciona producto --</option>'+productos.map(p=>`<option value="${p.id}">${p.nombre}</option>`).join('');
  sel.value=cur;
}
function registrarMovimiento(){
  const pid=document.getElementById('movProducto').value;
  const idx=productos.findIndex(p=>p.id==pid);
  if(idx<0){alert('Selecciona un producto.');return;}
  const cant=document.getElementById('movCantidad').value.trim();
  if(!cant){alert('Introduce una cantidad.');return;}
  aplicarMov(idx,document.getElementById('movTipo').value,cant,document.getElementById('movMotivo').value.trim(),document.getElementById('movResponsable').value.trim());
  ['movCantidad','movMotivo'].forEach(id=>document.getElementById(id).value='');
}
function aplicarMov(idx,tipo,cant,motivo,resp){
  const p=productos[idx];
  movimientos.unshift({fecha:new Date().toLocaleString('es-ES'),producto:p.nombre,tipo,cantidad:cant,unidad:p.unidad,motivo:motivo||'-',resp:resp||'-'});
  saveData();renderMovimientos();
}
function renderMovimientos(){
  const el=document.getElementById('listMovimientos');
  if(!movimientos.length){el.innerHTML='<div class="empty">Sin movimientos.</div>';return;}
  el.innerHTML=movimientos.map(m=>`<div class="mov-item ${m.tipo==='entrada'?'mov-entry':'mov-exit'}">
    <strong>${m.tipo==='entrada'?'📥 ENTRADA':'📤 SALIDA'}</strong> — ${m.producto}: <strong>${m.cantidad} ${m.unidad}</strong>${m.motivo!=='-'?' · '+m.motivo:''}
    <div class="mov-date">${m.fecha}${m.resp!=='-'?' · '+m.resp:''}</div>
  </div>`).join('');
}

// NOTAS
function addNota(){
  const titulo=document.getElementById('notaTitulo').value.trim();
  const texto=document.getElementById('notaTexto').value.trim();
  if(!titulo&&!texto){alert('Escribe algo.');return;}
  notas.unshift({id:uid(),titulo:titulo||'Sin titulo',texto,fecha:new Date().toLocaleString('es-ES'),usuario:usuarioActual?usuarioActual.nombre:'-'});
  document.getElementById('notaTitulo').value='';document.getElementById('notaTexto').value='';
  saveData();renderNotas();
}
function renderNotas(){
  const el=document.getElementById('listNotas');
  if(!notas.length){el.innerHTML='<div class="empty">No hay notas.</div>';return;}
  el.innerHTML=notas.map((n,i)=>`<div class="nota-item">
    <div class="nota-titulo">${esc(n.titulo)}</div>
    <div class="nota-texto">${esc(n.texto)}</div>
    <div class="nota-meta">${n.fecha} · ${n.usuario}</div>
    <button class="btn-sm btn-danger" style="margin-top:8px" onclick="borrarNota(${i})">Eliminar</button>
  </div>`).join('');
}
function borrarNota(i){if(confirm('Eliminar nota?')){notas.splice(i,1);saveData();renderNotas();}}

// WHATSAPP
function generarTextoInventario(prods,semana,resp){
  const fecha=hoy.toLocaleDateString('es-ES');
  let txt='🐟 *DON BACALAO - ALMACEN SALAMANCA*\n📅 '+semana+'\n👤 '+resp+'\n📆 '+fecha+'\n'+'─'.repeat(28)+'\n';
  const cats=[...new Set(prods.map(p=>p.categoria))];
  cats.forEach(cat=>{
    const items=prods.filter(p=>p.categoria===cat&&p.cantidad);
    if(!items.length)return;
    txt+='\n*'+cat.toUpperCase()+'*\n';
    items.forEach((p,i)=>txt+=(i+1)+'. '+p.nombre+': *'+p.cantidad+' '+p.unidad+'*'+(p.bloqueado?' ⛔':'')+'\n');
  });
  txt+='\n'+'─'.repeat(28)+'\nTotal con cantidad: '+prods.filter(p=>p.cantidad).length+' articulos';
  return txt;
}
function compartirWhatsApp(){
  const semana=document.getElementById('semanaLabel').value||'Inventario';
  const resp=document.getElementById('responsableLabel').value||'-';
  window.open('https://wa.me/?text='+encodeURIComponent(generarTextoInventario(productos,semana,resp)),'_blank');
}
function whatsappHistorial(i){
  const h=historial[i];
  window.open('https://wa.me/?text='+encodeURIComponent(generarTextoInventario(h.productos,h.semana,h.resp)),'_blank');
}

    items.forEach((p,i)=>{const bg=rn%2===0?'#fff':'#f7fafc';rn++;rows+=`<tr style="background:${bg}"><td style="padding:4px 7px;border-bottom:1px solid #e8e8e8;font-size:10px;color:#666">${i+1}</td><td style="padding:4px 7px;border-bottom:1px solid #e8e8e8;font-size:10.5px">${p.nombre||''}</td><td style="padding:4px 7px;border-bottom:1px solid #e8e8e8;font-size:10px;color:#555">${p.unidad||''}</td><td style="padding:4px 7px;border-bottom:1px solid #e8e8e8;font-size:10.5px;font-weight:bold;text-align:center">${p.cantidad||''}</td><td style="padding:4px 7px;border-bottom:1px solid #e8e8e8;font-size:10px;color:${p.bloqueado?'#c53030':'#276749'};font-weight:bold">${p.bloqueado?'BLOQUEADO':'OK'}</td></tr>`;});
  });
  const c=`<!DOCTYPE html><html><head><meta charset="UTF-8"><title>Estupaje Don Bacalao</title><style>@page{margin:15mm}body{font-family:Arial,sans-serif;margin:0}table{width:100%;border-collapse:collapse}th{background:#2d4a6e;color:white;padding:5px 8px;font-size:10px;text-align:left}</style></head><body><div style="background:#1e3a5f;color:white;padding:12px 16px;margin-bottom:12px"><div style="font-size:16px;font-weight:bold">DON BACALAO - ALMACEN DE SALAMANCA</div><div style="font-size:10px;margin-top:4px;opacity:.85">${semana}${responsable?' | Responsable: '+responsable:''} | Fecha: ${fecha}</div></div><table><thead><tr><th style="width:30px">#</th><th>Articulo</th><th style="width:90px">Unidad</th><th style="width:80px;text-align:center">Cantidad</th><th style="width:80px">Estado</th></tr></thead><tbody>${rows}</tbody></table><div style="margin-top:14px;font-size:9px;color:#aaa;border-top:1px solid #eee;padding-top:6px">Total: ${prods.length} articulos | Don Bacalao - Almacen de Salamanca</div></body></html>`;
  const blob=new Blob([c],{type:'text/html;charset=utf-8'});
  const url=URL.createObjectURL(blob);
  const a=document.createElement('a');a.href=url;a.download='estupaje_'+fecha.replace(/\//g,'-')+'.html';
  document.body.appendChild(a);a.click();document.body.removeChild(a);
  setTimeout(()=>URL.revokeObjectURL(url),1000);
  setTimeout(()=>alert('Descargado. Abrelo y pulsa Ctrl+P para guardar como PDF.'),300);
}

function importCSV(e){
  const file=e.target.files[0];if(!file)return;
  const reader=new FileReader();
  reader.onload=ev=>{
    const lines=ev.target.result.split(/\r?\n/).filter(l=>l.trim());let added=0;
    lines.forEach((line,idx)=>{const sep=line.includes(';')?';':',';const c=line.split(sep).map(x=>x.trim().replace(/^"|"$/g,''));if(idx===0&&isNaN(c[1]))return;const[nombre,cantidad,unidad,,categoria]=c;if(nombre){productos.push({id:uid(),nombre,cantidad:cantidad||'',unidad:unidad||'ud',categoria:categoria||'Extra',bloqueado:false,fijo:false,nota:''});added++;}});
    saveData();renderAll();alert(added+' productos importados.');
  };
  reader.readAsText(file,'UTF-8');e.target.value='';
}

function getWeekNum(){const d=new Date(Date.UTC(hoy.getFullYear(),hoy.getMonth(),hoy.getDate()));d.setUTCDate(d.getUTCDate()+4-(d.getUTCDay()||7));return Math.ceil((((d-new Date(Date.UTC(d.getUTCFullYear(),0,1)))/86400000)+1)/7);}
function esc(s){return(s||'').replace(/&/g,'&amp;').replace(/"/g,'&quot;').replace(/</g,'&lt;');}
